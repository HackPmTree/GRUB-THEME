# ---------------------------------------------------------------------------
# РАБОЧИЕ КНОПКИ ПАНЕЛИ УПРАВЛЕНИЯ
#   В GRUB «рабочими» кнопками могут быть только ПУНКТЫ МЕНЮ: Enter на
#   пункте выполняет действие. Действия задаются в /boot/grub/grub.cfg
#   встроенными переменными-«якорями» (@grub_dir, @bootdir, @rootdir):
#
#     menuentry 'Настройки (терминал GRUB)' --id=settings { set root='(hd0)' ; terminal_input console ; terminal_output gfxterm }
#     submenu 'Загрузка / Параметры' --id=bootcfg { ...пункты загрузки... }
#     menuentry 'Выключить' --id=poweroff { halt }
#     menuentry 'Перезагрузить' --id=reboot { reboot }
#
#   (готовые строки — в README.md; там же скрипт-хук /etc/grub.d/40_custom)
#   Иконки пунктов: icons/settings.png, icons/gear.png, icons/uefi.png.
#   Плашки btn-*.png / btn-*-sel.png используются в превью и как стиль
#   подсветки; нормальный/выбранный вид кнопки = color/selected_color выше.
# ---------------------------------------------------------------------------
