# ============================================================================
#  ТЕМА GRUB «СССР» — красно-золотая, конструктивизм 1930–1950-х

#  Палитра: #CC0000 красный · #FFD700 золотой · #8B0000 тёмно-красный · #000000 чёрный
#  Все текстовые элементы — на русском языке (шрифт DejaVu Sans с кириллицей).
# ============================================================================

# ---------------------------------------------------------------------------
# Разрешения. GRUB сам масштабирует desktop-image под текущий видеорежим
# (1920x1080, 1366x768, 1280x720, 2560x1440, 3840x2160 — все варианты лежат
# в корне темы: background-1366x768.png и т.д.). Координаты ниже заданы
# для канонических 1920x1080; при меньших разрешениях GRUB пересчитывает
# их пропорционально размеру экрана.
# ---------------------------------------------------------------------------

desktop-image: "background.png"
# desktop-color: "#000000"   # запасной цвет, если изображение не загрузится

# ---------------------------------------------------------------------------
# ШРИФТЫ (кириллица обязательна!)
#   theme/fonts/dejavu-sans-16.pf2  — основной (меню)
#   theme/fonts/dejavu-sans-18.pf2  — заголовок/таймер
# Собираются командой:
#   grub-mkfont -o fonts/dejavu-sans-16.pf2 --size=16 /usr/share/fonts/truetype/dejavu/DejaVuSans.ttf
# ---------------------------------------------------------------------------

+ global_theme {
    font = "fonts/dejavu-sans-16.pf2"
    terminal-font = "fonts/dejavu-sans-16.pf2"
}

# ---------------------------------------------------------------------------

    icons_on = "icons/emblem-64.png"
    align = "center"
    top = 2%
    left = 0
    width = 100%
    height = 48
}

# ---------------------------------------------------------------------------
# МЕНЮ ВЫБОРА ОСЕЙ
# ---------------------------------------------------------------------------
+ menu_panel {
    + item {
        height = 42
        width = 100%
        spacing = 6
        icon_width = 40
        icon_height = 40
        item_icon_space = 14
        item_padding = 18
        item_alignment = "left"

        color = "#FFF6CD"            # обычный пункт — светло-жёлтый (кремовый)
        selected_color = "#FFD700"   # выделенный пункт — золотой
        # фон выделения задаётся рамкой selected_item.png (см. ниже),
        # поэтому color_selected оставляем золотым по тёмно-красному:
        selected_image = "selected_item.png"

        # Подсказка под меню — тоже по-русски
        + selected_item {
            item_font = "fonts/dejavu-sans-16.pf2"
        }
    }

    align = "left"
    top = 34%
    left = 5%
    width = 46%
    height = 46%
    border_color = "#8B0000"
    bg_color = "#000000AA"           # полупрозрачная «чёрная доска» под списком
    # (некоторые сборки GRUB игнорируют альфа-канал — тогда будет #000000)
}

# ---------------------------------------------------------------------------
# КАНОНИЧЕСКИЕ ПУНКТЫ МЕНЮ (русские названия; реальные пункты берутся
# из grub.cfg, здесь задаются только стиль и иконки через match)
# ---------------------------------------------------------------------------
# Примеры раскладки пунктов (для документации/скриптов генерации cfg):
#   1. Арх Линукс (Ядро Linux)              → icons/arch.png
#   2. Арх Линукс (Запуск по умолчанию)     → icons/star.png
#   3. Манжаро Линукс (Ядро 6.x)            → icons/manjaro.png
#   4. Дополнительные параметры ядра        → icons/gear.png
#   5. Загрузка в текстовом режиме (console) → icons/terminal.png
#   6. Проверка оперативной памяти          → icons/memtest.png
#   7. Другая операционная система (Windows) → icons/windows.png
#   8. Перезагрузка / Выключить компьютер    → icons/uefi.png

+ menu {
    style = "menu_item"
    # Прокрутка списка:
    + slider {
        width = 18
        top = 0
        right = 0
        height = 100%
        bg_image = "slider_bg.png"
        + thumb {
            image = "slider_thumb.png"   # золотая звезда на красном круге
            width = 18
            height = 18
        }
    }
}

# ---------------------------------------------------------------------------
# ТАЙМЕР ОБРАТНОГО ОТСЧЁТА В СОВЕТСКОМ СТИЛЕ
#   Статичная плашка countdown_frame.png (знамя + звезда + лавр) лежит

#   вместо неё используется встроенный виджет progress (полоса знамённая):
#   она убывает от золотой к красной по мере истечения timeout.
# ---------------------------------------------------------------------------
+ message {
    bg_image = "countdown_frame.png"
    bg_color = "#8B0000"
    fg_color = "#FFD700"
    font = "fonts/dejavu-sans-18.pf2"

    top = 84%
    right = 3%
    width = 280
    height = 120
    text = ""
}

+ timeout_style {
    # Строка «Осталось: N сек.» рисуется поверх плашки таймера

    font = "fonts/dejavu-sans-18.pf2"
    align = "bottom_right"

    right = 3%
    width = 280
    height = 120
}
